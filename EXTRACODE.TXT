# ============================================================
# 1. IMPORTS
# ============================================================
from typing import TypedDict, List, Any
from langgraph.graph import StateGraph

# You already defined these earlier:
# - llm_openrouter (OpenAILike)
# - USE_OPENROUTER
# - query_engine (RetrieverQueryEngine)
# So we simply reuse them here.

# Select LLM (OpenRouter only for this version)
llm = llm_openrouter


# ============================================================
# 2. SHARED STATE
# ============================================================
class AgentState(TypedDict):
    user_query: str
    supervisor_decision: str
    rag_answer: str
    calc_answer: str
    code_answer: str
    memory: List[str]
    plan: str
    final: str


# ============================================================
# 3. MEMORY NODE
# ============================================================
def memory_node(state: AgentState) -> AgentState:
    updated_memory = state["memory"] + [state["user_query"]]
    return {**state, "memory": updated_memory}


# ============================================================
# 4. PLANNING AGENT
# ============================================================
def planning_agent(state: AgentState) -> AgentState:
    prompt = f"""
    You are the Planning Agent.

    Your job:
    - Look at the user query and conversation memory.
    - Decide which capabilities are needed: RAG, CALC, CODE.
    - Produce a short step-by-step plan.

    Memory:
    {state['memory']}

    User query:
    {state['user_query']}

    Respond with a concise plan.
    """

    resp = llm.complete(prompt)
    plan = getattr(resp, "text", str(resp)).strip()

    return {**state, "plan": plan}


# ============================================================
# 5. SUPERVISOR AGENT
# ============================================================
def supervisor_agent(state: AgentState) -> AgentState:
    prompt = f"""
    You are the Supervisor Agent.

    You have a plan from the Planning Agent:
    {state['plan']}

    Decide which SINGLE agent should answer the user query NOW:
    - RAG
    - CALC
    - CODE

    User query: {state['user_query']}

    Respond with ONLY one word: RAG, CALC, or CODE.
    """

    resp = llm.complete(prompt)
    decision = getattr(resp, "text", str(resp)).strip().upper()

    return {**state, "supervisor_decision": decision}


# ============================================================
# 6. RAG AGENT (USING YOUR query_engine)
# ============================================================
def rag_agent(state: AgentState) -> AgentState:
    query = state["user_query"]

    # LlamaIndex RAG pipeline
    resp = query_engine.query(query)
    answer = str(resp)

    return {**state, "rag_answer": answer}


# ============================================================
# 7. CALCULATOR AGENT
# ============================================================
def calc_agent(state: AgentState) -> AgentState:
    try:
        result = eval(state["user_query"])
    except Exception as e:
        result = f"Invalid expression: {e}"

    return {**state, "calc_answer": str(result)}


# ============================================================
# 8. CODE AGENT
# ============================================================
def code_agent(state: AgentState) -> AgentState:
    code = state["user_query"]
    local_vars: dict[str, Any] = {}

    try:
        exec(code, {}, local_vars)
        result = str(local_vars)
    except Exception as e:
        result = str(e)

    return {**state, "code_answer": result}


# ============================================================
# 9. PARALLEL AGENTS NODE
# ============================================================
def parallel_agents(state: AgentState) -> AgentState:
    rag = rag_agent(state)
    calc = calc_agent(state)
    code = code_agent(state)

    return {
        **state,
        "rag_answer": rag["rag_answer"],
        "calc_answer": calc["calc_answer"],
        "code_answer": code["code_answer"],
    }


# ============================================================
# 10. FINAL ANSWER NODE (FIXED)
# ============================================================
def final_answer(state: AgentState) -> AgentState:
    decision = state["supervisor_decision"]

    if decision == "RAG":
        answer = state["rag_answer"]
    elif decision == "CALC":
        answer = state["calc_answer"]
    elif decision == "CODE":
        answer = state["code_answer"]
    else:
        answer = "Supervisor could not decide."

    return {**state, "final": answer}


# ============================================================
# 11. BUILD LANGGRAPH WORKFLOW
# ============================================================
workflow = StateGraph(AgentState)

workflow.add_node("memory", memory_node)
workflow.add_node("planner", planning_agent)
workflow.add_node("supervisor", supervisor_agent)
workflow.add_node("parallel", parallel_agents)
workflow.add_node("final", final_answer)

workflow.set_entry_point("memory")

workflow.add_edge("memory", "planner")
workflow.add_edge("planner", "supervisor")
workflow.add_edge("supervisor", "parallel")
workflow.add_edge("parallel", "final")

app = workflow.compile()


# ============================================================
# 12. RUN THE SYSTEM
# ============================================================
result = app.invoke({
    "user_query": "Explain how RAG works and then compute 12 * 9.",
    "supervisor_decision": "",
    "rag_answer": "",
    "calc_answer": "",
    "code_answer": "",
    "memory": [],
    "plan": "",
    "final": ""
})

print("\n================ PLAN ================\n")
print(result["plan"])

print("\n================ FINAL ANSWER ================\n")
print(result["final"])

print("\n================ MEMORY ================\n")
print(result["memory"])
